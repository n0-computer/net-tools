1//! Minimal rtnetlink client for netwatch.
2//!
3//! Covers exactly the subset of the netlink route protocol that netwatch
4//! uses on Linux and Android:
5//!
6//! - dumping links, addresses and routes ([`Connection`] for blocking
7//!   callers, [`AsyncConnection`] inside tokio),
8//! - looking up a single link by interface index,
9//! - listening to rtnetlink multicast groups for change events
10//!   ([`EventSocket`]).
11//!
12//! Messages are parsed into the small typed structs in this crate
13//! ([`LinkMessage`], [`AddressMessage`], [`RouteMessage`]); attributes we do
14//! not use are skipped. On every platform other than Linux and Android the
15//! crate compiles to nothing.
16#![cfg(any(target_os = "linux", target_os = "android"))]
17

29/// Errors surfaced by the netlink socket wrappers.
30#[stack_error(derive, add_meta, from_sources, std_sources)]
31#[non_exhaustive]
32pub enum Error {
33    /// A socket operation failed.
34    #[error("IO")]
35    Io { source: std::io::Error },
36    /// The kernel replied with an `NLMSG_ERROR` message.
37    ///
38    /// `code` is a negative errno value.
39    #[error("netlink error message: code {code}")]
40    ErrorMessage { code: i32 },
41    /// The kernel dropped events because the socket buffer overran.
42    ///
43    /// Subscribers should resynchronize by re-reading the state they care
44    /// about.
45    #[error("netlink socket buffer overrun")]
46    Overrun {},
47    /// The kernel signaled the end of a multipart message on an event
48    /// socket.
49    ///
50    /// Subscribers should treat this like a lost connection and
51    /// resubscribe.
52    #[error("end of multipart message")]
53    Done {},
54    /// A datagram did not fit the receive buffer and was truncated.
55    #[error("truncated netlink datagram")]
56    Truncated {},
57}

59/// Returns the `sockaddr_nl` group mask bit for an rtnetlink multicast
60/// group.
61///
62/// Only the first 31 groups can be subscribed through the bind mask; all
63/// `RTNLGRP_*` groups netwatch uses fall in that range.
64///
65/// # Panics
66///
67/// Panics when `group` is larger than 31.
68pub const fn group_flag(group: u32) -> u32 {
69    assert!(group <= 31, "group not reachable via the bind mask");
70    if group == 0 { 0 } else { 1 << (group - 1) }
71}
