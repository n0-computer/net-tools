25/// Receive buffer size for a single datagram.
26///
27/// The kernel caps rtnetlink dump datagrams well below this (32k), so a
28/// larger buffer only wastes memory.
29const RECV_BUF_SIZE: usize = 64 * 1024;
30
31/// Deadline for a whole dump.
32///
33/// When the deadline passes, the messages collected so far are returned:
34/// enumeration should degrade rather than fail when the kernel drops a
35/// datagram.
36const DUMP_TIMEOUT: Duration = Duration::from_secs(2);
37
38/// Builds the netlink socket address for the given multicast group mask.
39///
40/// A zero mask addresses the kernel for request/response use.
41fn netlink_addr(groups: u32) -> SockAddr {
42    // SAFETY: an all-zero sockaddr_nl is valid; only the family and the
43    // group mask need real values (pid zero lets the kernel assign one).
44    // The storage is larger than sockaddr_nl and the length says so.
45    unsafe {
46        let mut storage = socket2::SockAddrStorage::zeroed();
47        let addr = std::ptr::from_mut(&mut storage).cast::<libc::sockaddr_nl>();
48        (*addr).nl_family = libc::AF_NETLINK as libc::sa_family_t;
49        (*addr).nl_groups = groups;
50        SockAddr::new(
51            storage,
52            std::mem::size_of::<libc::sockaddr_nl>() as libc::socklen_t,
53        )
54    }
55}

57/// A `NETLINK_ROUTE` socket.
58#[derive(Debug)]
59struct NetlinkSocket {
60    socket: Socket,
61}
62
63impl NetlinkSocket {
64    /// Opens the socket, subscribed to the multicast groups in `groups`
65    /// (zero for request/response use).
66    ///
67    /// Blocking receives use a per-call timeout; async users flip the
68    /// socket to non-blocking and drive it through an [`AsyncFd`].
69    fn new(groups: u32, nonblocking: bool) -> io::Result<Self> {
70        let socket = Socket::new(
71            Domain::from(libc::AF_NETLINK),
72            Type::DGRAM,
73            Some(Protocol::from(libc::NETLINK_ROUTE)),
74        )?;
75        socket.set_nonblocking(nonblocking)?;
76
77        // On Android 11+ SELinux denies bind on netlink route sockets for
78        // apps; the kernel auto-binds on the first send instead. Group
79        // subscription requires the bind and is not used on Android
80        // (netmon is a no-op there).
81        let bind = groups != 0 || cfg!(not(target_os = "android"));
82        if bind {
83            socket.bind(&netlink_addr(groups))?;
84        }
85        Ok(Self { socket })
86    }
87
88    /// Sends a request datagram to the kernel.
89    fn send_request(&self, buf: &[u8]) -> io::Result<()> {
90        let sent = self.socket.send_to(buf, &netlink_addr(0))?;
91        // Datagram sockets send whole messages; a short send cannot happen.
92        debug_assert_eq!(sent, buf.len());
93        Ok(())
94    }
95
96    /// Receives one datagram.
97    ///
98    /// Returns the datagram's true length, which exceeds `buf.len()` when
99    /// the datagram was truncated (`MSG_TRUNC`).
100    fn recv(&self, buf: &mut [u8]) -> io::Result<usize> {
101        // SAFETY: MaybeUninit<u8> has the same layout as u8, and the
102        // receive only writes initialized bytes into the buffer.
103        let buf = unsafe { &mut *(std::ptr::from_mut::<[u8]>(buf) as *mut [MaybeUninit<u8>]) };
104        self.socket.recv_with_flags(buf, libc::MSG_TRUNC)
105    }
106
107    /// Blocking receive of one datagram, bounded by `deadline`.
108    ///
109    /// Returns `None` when the deadline passes first.
110    fn recv_deadline(&self, buf: &mut [u8], deadline: Instant) -> io::Result<Option<usize>> {
111        loop {
112            let Some(remaining) = deadline.checked_duration_since(Instant::now()) else {
113                return Ok(None);
114            };
115            // A zero timeout would mean "block forever".
116            self.socket
117                .set_read_timeout(Some(remaining.max(Duration::from_millis(1))))?;
118            match self.recv(buf) {
119                Ok(len) => return Ok(Some(len)),
120                Err(err) if err.kind() == io::ErrorKind::Interrupted => {}
121                Err(err)
122                    if matches!(
123                        err.kind(),
124                        io::ErrorKind::WouldBlock | io::ErrorKind::TimedOut
125                    ) =>
126                {
127                    return Ok(None);
128                }
129                Err(err) => return Err(err),
130            }
131        }
132    }
133}

141/// Maps a receive error, turning `ENOBUFS` into [`Error::Overrun`].
142fn map_recv_err(err: io::Error) -> Error {
143    if err.raw_os_error() == Some(libc::ENOBUFS) {
144        e!(Error::Overrun)
145    } else {
146        e!(Error::Io, err)
147    }
148}

150/// Collects the frames of one dump, keyed by sequence number.
151#[derive(Debug, Default)]
152struct DumpCollector {
153    messages: Vec<Message>,
154    done: bool,
155}
156
157impl DumpCollector {
158    fn push_datagram(&mut self, seq: u32, datagram: &[u8]) -> Result<(), Error> {
159        for frame in message::parse_frames(datagram) {
160            match frame {
161                Frame::Message { seq: s, message } if s == seq => self.messages.push(message),
162                Frame::Done { seq: s } if s == seq => {
163                    self.done = true;
164                    return Ok(());
165                }
166                Frame::Error { seq: s, code } if s == seq && code != 0 => {
167                    return Err(e!(Error::ErrorMessage { code }));
168                }
169                // Acks (code zero), foreign sequence numbers and skipped
170                // frames are ignored.
171                _ => {}
172            }
173        }
174        Ok(())
175    }
176}

178fn filter_links(messages: Vec<Message>) -> Vec<LinkMessage> {
179    messages
180        .into_iter()
181        .filter_map(|message| match message {
182            Message::NewLink(link) => Some(link),
183            _ => None,
184        })
185        .collect()
186}
187
188fn filter_addresses(messages: Vec<Message>) -> Vec<AddressMessage> {
189    messages
190        .into_iter()
191        .filter_map(|message| match message {
192            Message::NewAddress(address) => Some(address),
193            _ => None,
194        })
195        .collect()
196}

198fn filter_routes(messages: Vec<Message>) -> Vec<RouteMessage> {
199    messages
200        .into_iter()
201        .filter_map(|message| match message {
202            Message::NewRoute(route) => Some(route),
203            _ => None,
204        })
205        .collect()
206}

208/// A blocking request/response connection.
209///
210/// Dumps wait at most two seconds and return the messages received so far
211/// when the deadline passes.
212#[derive(Debug)]
213pub struct Connection {
214    socket: NetlinkSocket,
215    seq: u32,
216    buf: Vec<u8>,
217}
218
219impl Connection {
220    /// Opens a new connection.
221    pub fn new() -> Result<Self, Error> {
222        Ok(Self {
223            socket: NetlinkSocket::new(0, false)?,
224            seq: 0,
225            buf: vec![0; RECV_BUF_SIZE],
226        })
227    }
228
229    /// Dumps all links.
230    pub fn dump_links(&mut self) -> Result<Vec<LinkMessage>, Error> {
231        self.dump(dump_links_request).map(filter_links)
232    }
233
234    /// Dumps all addresses of both families.
235    pub fn dump_addresses(&mut self) -> Result<Vec<AddressMessage>, Error> {
236        self.dump(dump_addresses_request).map(filter_addresses)
237    }
238
239    /// Dumps the routes of the given family.
240    pub fn dump_routes(&mut self, family: RouteFamily) -> Result<Vec<RouteMessage>, Error> {
241        self.dump(|seq| dump_routes_request(seq, family))
242            .map(filter_routes)
243    }
244
245    fn next_seq(&mut self) -> u32 {
246        self.seq = self.seq.wrapping_add(1);
247        self.seq
248    }
249
250    fn dump(&mut self, build: impl FnOnce(u32) -> Vec<u8>) -> Result<Vec<Message>, Error> {
251        let seq = self.next_seq();
252        self.socket.send_request(&build(seq))?;
253        let deadline = Instant::now() + DUMP_TIMEOUT;
254        let mut collector = DumpCollector::default();
255        while !collector.done {
256            match self.socket.recv_deadline(&mut self.buf, deadline) {
257                Ok(None) => {
258                    warn!("netlink dump timed out, returning partial result");
259                    break;
260                }
261                Ok(Some(len)) if len > self.buf.len() => return Err(e!(Error::Truncated)),
262                Ok(Some(len)) => collector.push_datagram(seq, &self.buf[..len])?,
263                Err(err) => return Err(map_recv_err(err)),
264            }
265        }
266        Ok(collector.messages)
267    }
268}

270/// An async request/response connection.
271///
272/// The async twin of [`Connection`], with the same dump semantics.
273#[derive(Debug)]
274pub struct AsyncConnection {
275    socket: AsyncFd<NetlinkSocket>,
276    seq: u32,
277    buf: Vec<u8>,
278}
279
280impl AsyncConnection {
281    /// Opens a new connection.
282    ///
283    /// Must be called from within a tokio runtime.
284    pub fn new() -> Result<Self, Error> {
285        let socket = NetlinkSocket::new(0, true)?;
286        Ok(Self {
287            socket: AsyncFd::new(socket).map_err(|err| e!(Error::Io, err))?,
288            seq: 0,
289            buf: vec![0; RECV_BUF_SIZE],
290        })
291    }
292
293    /// Dumps all links.
294    pub async fn dump_links(&mut self) -> Result<Vec<LinkMessage>, Error> {
295        self.dump(dump_links_request).await.map(filter_links)
296    }
297
298    /// Dumps all addresses of both families.
299    pub async fn dump_addresses(&mut self) -> Result<Vec<AddressMessage>, Error> {
300        self.dump(dump_addresses_request)
301            .await
302            .map(filter_addresses)
303    }
304
305    /// Dumps the routes of the given family.
306    pub async fn dump_routes(&mut self, family: RouteFamily) -> Result<Vec<RouteMessage>, Error> {
307        self.dump(|seq| dump_routes_request(seq, family))
308            .await
309            .map(filter_routes)
310    }
311
312    /// Requests a single link by interface index.
313    ///
314    /// Returns `None` when the kernel does not answer within the dump
315    /// deadline.
316    pub async fn get_link_by_index(&mut self, index: u32) -> Result<Option<LinkMessage>, Error> {
317        let seq = self.next_seq();
318        self.socket
319            .get_ref()
320            .send_request(&get_link_request(seq, index))?;
321        let deadline = tokio::time::Instant::now() + DUMP_TIMEOUT;
322        loop {
323            let Some(len) = self.recv_datagram(deadline).await? else {
324                return Ok(None);
325            };
326            for frame in message::parse_frames(&self.buf[..len]) {
327                match frame {
328                    Frame::Message {
329                        seq: s,
330                        message: Message::NewLink(link),
331                    } if s == seq => return Ok(Some(link)),
332                    Frame::Error { seq: s, code } if s == seq && code != 0 => {
333                        return Err(e!(Error::ErrorMessage { code }));
334                    }
335                    Frame::Done { seq: s } if s == seq => return Ok(None),
336                    _ => {}
337                }
338            }
339        }
340    }
341
342    fn next_seq(&mut self) -> u32 {
343        self.seq = self.seq.wrapping_add(1);
344        self.seq
345    }
346
347    async fn dump(&mut self, build: impl FnOnce(u32) -> Vec<u8>) -> Result<Vec<Message>, Error> {
348        let seq = self.next_seq();
349        self.socket.get_ref().send_request(&build(seq))?;
350        let deadline = tokio::time::Instant::now() + DUMP_TIMEOUT;
351        let mut collector = DumpCollector::default();
352        while !collector.done {
353            let Some(len) = self.recv_datagram(deadline).await? else {
354                warn!("netlink dump timed out, returning partial result");
355                break;
356            };
357            collector.push_datagram(seq, &self.buf[..len])?;
358        }
359        Ok(collector.messages)
360    }
361
362    /// Receives one datagram into the connection buffer.
363    ///
364    /// Returns its length, or `None` when `deadline` passes first.
365    async fn recv_datagram(
366        &mut self,
367        deadline: tokio::time::Instant,
368    ) -> Result<Option<usize>, Error> {
369        let Self { socket, buf, .. } = self;
370        loop {
371            let Ok(guard) = tokio::time::timeout_at(deadline, socket.readable()).await else {
372                return Ok(None);
373            };
374            let mut guard = guard.map_err(|err| e!(Error::Io, err))?;
375            match guard.try_io(|socket| socket.get_ref().recv(buf)) {
376                Ok(Ok(len)) if len > buf.len() => return Err(e!(Error::Truncated)),
377                Ok(Ok(len)) => return Ok(Some(len)),
378                Ok(Err(err)) => return Err(map_recv_err(err)),
379                // Spurious readiness: wait again.
380                Err(_would_block) => {}
381            }
382        }
383    }
384}

386/// A pending event parsed from a received datagram.
387#[derive(Debug)]
388enum PendingEvent {
389    Message(Message),
390    Done,
391    Error { code: i32 },
392}
393
394/// A socket subscribed to rtnetlink multicast groups.
395///
396/// Build the `groups` mask from `RTNLGRP_*` values with
397/// [`group_flag`](crate::group_flag).
398#[derive(Debug)]
399pub struct EventSocket {
400    socket: AsyncFd<NetlinkSocket>,
401    buf: Vec<u8>,
402    pending: VecDeque<PendingEvent>,
403}
404
405impl EventSocket {
406    /// Subscribes to the multicast groups in the `groups` bind mask.
407    ///
408    /// Must be called from within a tokio runtime.
409    pub fn subscribe(groups: u32) -> Result<Self, Error> {
410        let socket = NetlinkSocket::new(groups, true)?;
411        Ok(Self {
412            socket: AsyncFd::new(socket).map_err(|err| e!(Error::Io, err))?,
413            buf: vec![0; RECV_BUF_SIZE],
414            pending: VecDeque::new(),
415        })
416    }
417
418    /// Waits for the next event message.
419    ///
420    /// # Errors
421    ///
422    /// - [`Error::ErrorMessage`]: the kernel sent an error frame; the
423    ///   socket remains usable and `next` can be called again.
424    /// - [`Error::Overrun`], [`Error::Done`], [`Error::Truncated`],
425    ///   [`Error::Io`]: events may have been lost; the caller should drop
426    ///   the socket, resubscribe, and re-read the state it watches.
427    pub async fn next(&mut self) -> Result<Message, Error> {
428        loop {
429            match self.pending.pop_front() {
430                Some(PendingEvent::Message(message)) => return Ok(message),
431                Some(PendingEvent::Done) => return Err(e!(Error::Done)),
432                Some(PendingEvent::Error { code }) => return Err(e!(Error::ErrorMessage { code })),
433                None => {}
434            }
435            let Self {
436                socket,
437                buf,
438                pending,
439            } = self;
440            let mut guard = socket.readable().await.map_err(|err| e!(Error::Io, err))?;
441            match guard.try_io(|socket| socket.get_ref().recv(buf)) {
442                Ok(Ok(len)) if len > buf.len() => return Err(e!(Error::Truncated)),
443                Ok(Ok(len)) => {
444                    for frame in message::parse_frames(&buf[..len]) {
445                        match frame {
446                            Frame::Message { message, .. } => {
447                                pending.push_back(PendingEvent::Message(message));
448                            }
449                            Frame::Done { .. } => pending.push_back(PendingEvent::Done),
450                            Frame::Error { code, .. } => {
451                                pending.push_back(PendingEvent::Error { code });
452                            }
453                            Frame::Skip => {}
454                        }
455                    }
456                }
457                Ok(Err(err)) => return Err(map_recv_err(err)),
458                // Spurious readiness: wait again.
459                Err(_would_block) => {}
460            }
461        }
462    }
463}

469    fn loopback_link(links: &[LinkMessage]) -> &LinkMessage {
470        links
471            .iter()
472            .find(|link| link.name.as_deref() == Some("lo"))
473            .expect("no loopback link")
