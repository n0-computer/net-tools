1//! Typed rtnetlink messages and requests.
2//!
3//! Only the message families and attributes netwatch consumes are parsed;
4//! everything else is preserved as [`Message::Other`] or skipped.

10// Attribute types not exposed by the libc crate.
11const IFLA_ADDRESS: u16 = 1;
12const IFLA_IFNAME: u16 = 3;
13const IFA_ADDRESS: u16 = 1;
14const IFA_LOCAL: u16 = 2;
15const IFA_FLAGS: u16 = 8;
16const RTA_DST: u16 = 1;
17const RTA_OIF: u16 = 4;
18const RTA_GATEWAY: u16 = 5;
19const RTA_TABLE: u16 = 15;

21/// Address family selector for route dumps.
22#[derive(Debug, Clone, Copy, PartialEq, Eq)]
23pub enum RouteFamily {
24    /// Dump routes of every family.
25    Unspec,
26    /// Dump IPv4 routes.
27    Ipv4,
28    /// Dump IPv6 routes.
29    Ipv6,
30}
31
32impl RouteFamily {
33    fn family(self) -> u8 {
34        match self {
35            RouteFamily::Unspec => libc::AF_UNSPEC as u8,
36            RouteFamily::Ipv4 => libc::AF_INET as u8,
37            RouteFamily::Ipv6 => libc::AF_INET6 as u8,
38        }
39    }
40}

42/// A parsed `RTM_NEWLINK` or `RTM_DELLINK` message.
43#[derive(Debug, Clone)]
44pub struct LinkMessage {
45    /// Interface index from `ifinfomsg`.
46    pub index: u32,
47    /// Interface flags (`IFF_*`) from `ifinfomsg`.
48    pub flags: u32,
49    /// Interface name from `IFLA_IFNAME`.
50    pub name: Option<String>,
51    /// Hardware address bytes from `IFLA_ADDRESS`.
52    pub address: Option<Vec<u8>>,
53}
54
55impl LinkMessage {
56    /// Parses the message from an `ifinfomsg` payload.
57    ///
58    /// Wire layout: family `u8`, pad `u8`, type `u16`, index `i32`, flags
59    /// `u32`, change mask `u32`, then attributes.
60    fn parse(data: &[u8]) -> Option<Self> {
61        let index = wire::read_i32(data, 4)? as u32;
62        let flags = wire::read_u32(data, 8)?;
63        let mut name = None;
64        let mut address = None;
65        for (kind, payload) in AttrIter::new(data.get(16..)?) {
66            match kind {
67                IFLA_IFNAME => name = parse_string(payload),
68                IFLA_ADDRESS => address = Some(payload.to_vec()),
69                _ => {}
70            }
71        }
72        Some(Self {
73            index,
74            flags,
75            name,
76            address,
77        })
78    }
79}

81/// A parsed `RTM_NEWADDR` or `RTM_DELADDR` message.
82#[derive(Debug, Clone)]
83pub struct AddressMessage {
84    /// Address family (`AF_INET` or `AF_INET6`) from `ifaddrmsg`.
85    pub family: u8,
86    /// Prefix length of the address.
87    pub prefix_len: u8,
88    /// Address scope (`RT_SCOPE_*`) from `ifaddrmsg`.
89    pub scope: u8,
90    /// Interface index the address belongs to.
91    pub index: u32,
92    /// The address from `IFA_ADDRESS`.
93    ///
94    /// For IPv4 this is the peer address on point-to-point links and equal
95    /// to [`AddressMessage::local`] everywhere else; for IPv6 it is the
96    /// interface address.
97    pub address: Option<IpAddr>,
98    /// The local interface address from `IFA_LOCAL`, when present.
99    pub local: Option<IpAddr>,
100    /// Address flags (`IFA_F_*`) from the `IFA_FLAGS` attribute.
101    ///
102    /// `None` when the kernel did not send the attribute; use
103    /// [`AddressMessage::flags`] for the value with the header fallback
104    /// applied.
105    pub flags_attr: Option<u32>,
106    /// Address flags from the `ifaddrmsg` header byte.
107    ///
108    /// Truncated to eight bits; superseded by `flags_attr` when present.
109    pub header_flags: u8,
110}
111
112impl AddressMessage {
113    /// Parses the message from an `ifaddrmsg` payload.
114    ///
115    /// Wire layout: family `u8`, prefixlen `u8`, flags `u8`, scope `u8`,
116    /// index `u32`, then attributes.
117    fn parse(data: &[u8]) -> Option<Self> {
118        let family = *data.first()?;
119        let prefix_len = *data.get(1)?;
120        let header_flags = *data.get(2)?;
121        let scope = *data.get(3)?;
122        let index = wire::read_u32(data, 4)?;
123        let mut address = None;
124        let mut local = None;
125        let mut flags_attr = None;
126        for (kind, payload) in AttrIter::new(data.get(8..)?) {
127            match kind {
128                IFA_ADDRESS => address = parse_ip(family, payload),
129                IFA_LOCAL => local = parse_ip(family, payload),
130                IFA_FLAGS => flags_attr = wire::read_u32(payload, 0),
131                _ => {}
132            }
133        }
134        Some(Self {
135            family,
136            prefix_len,
137            scope,
138            index,
139            address,
140            local,
141            flags_attr,
142            header_flags,
143        })
144    }
145
146    /// The address flags (`IFA_F_*`), preferring the 32-bit `IFA_FLAGS`
147    /// attribute over the truncated header byte.
148    pub fn flags(&self) -> u32 {
149        self.flags_attr.unwrap_or(self.header_flags as u32)
150    }
151
152    /// The interface address, preferring `IFA_LOCAL` over `IFA_ADDRESS`.
153    ///
154    /// This matches what `ip addr` shows: for IPv4 the kernel puts the
155    /// interface address in `IFA_LOCAL` and the (peer) address in
156    /// `IFA_ADDRESS`; for IPv6 only `IFA_ADDRESS` is sent.
157    pub fn interface_address(&self) -> Option<IpAddr> {
158        self.local.or(self.address)
159    }
160}

162/// A parsed `RTM_NEWROUTE` or `RTM_DELROUTE` message.
163#[derive(Debug, Clone)]
164pub struct RouteMessage {
165    /// Address family (`AF_INET` or `AF_INET6`) from `rtmsg`.
166    pub family: u8,
167    /// Prefix length of the destination; zero for default routes.
168    pub dst_len: u8,
169    /// Routing table id from the `RTA_TABLE` attribute.
170    ///
171    /// `None` when the kernel did not send the attribute. The legacy
172    /// eight-bit `rtmsg` table byte is intentionally not exposed: netwatch
173    /// has always used only the attribute.
174    pub table: Option<u32>,
175    /// Route destination from `RTA_DST`; absent for default routes.
176    pub destination: Option<IpAddr>,
177    /// Gateway address from `RTA_GATEWAY`.
178    pub gateway: Option<IpAddr>,
179    /// Output interface index from `RTA_OIF`.
180    pub oif: Option<u32>,
181}
182
183impl RouteMessage {
184    /// Parses the message from an `rtmsg` payload.
185    ///
186    /// Wire layout: family, dst_len, src_len, tos, table, protocol, scope,
187    /// type (all `u8`), flags `u32`, then attributes.
188    fn parse(data: &[u8]) -> Option<Self> {
189        let family = *data.first()?;
190        let dst_len = *data.get(1)?;
191        let mut table = None;
192        let mut destination = None;
193        let mut gateway = None;
194        let mut oif = None;
195        for (kind, payload) in AttrIter::new(data.get(12..)?) {
196            match kind {
197                RTA_TABLE => table = wire::read_u32(payload, 0),
198                RTA_DST => destination = parse_ip(family, payload),
199                RTA_GATEWAY => gateway = parse_ip(family, payload),
200                RTA_OIF => oif = wire::read_u32(payload, 0),
201                _ => {}
202            }
203        }
204        Some(Self {
205            family,
206            dst_len,
207            table,
208            destination,
209            gateway,
210            oif,
211        })
212    }
213}

215/// A parsed rtnetlink message.
216///
217/// Rule messages carry no payload here because netwatch only reacts to
218/// their presence.
219#[derive(Debug, Clone)]
220pub enum Message {
221    /// A link was added or changed.
222    NewLink(LinkMessage),
223    /// A link was removed.
224    DelLink(LinkMessage),
225    /// An address was added or changed.
226    NewAddress(AddressMessage),
227    /// An address was removed.
228    DelAddress(AddressMessage),
229    /// A route was added or changed.
230    NewRoute(RouteMessage),
231    /// A route was removed.
232    DelRoute(RouteMessage),
233    /// A routing policy rule was added.
234    NewRule,
235    /// A routing policy rule was removed.
236    DelRule,
237    /// Any other rtnetlink message type.
238    Other {
239        /// The `nlmsghdr` message type.
240        kind: u16,
241    },
242}
243
244impl Message {
245    /// Parses a message payload for the given `nlmsghdr` type.
246    ///
247    /// Returns `None` when the payload is too short for its fixed header.
248    pub(crate) fn parse(kind: u16, payload: &[u8]) -> Option<Self> {
249        let message = match kind {
250            libc::RTM_NEWLINK => Message::NewLink(LinkMessage::parse(payload)?),
251            libc::RTM_DELLINK => Message::DelLink(LinkMessage::parse(payload)?),
252            libc::RTM_NEWADDR => Message::NewAddress(AddressMessage::parse(payload)?),
253            libc::RTM_DELADDR => Message::DelAddress(AddressMessage::parse(payload)?),
254            libc::RTM_NEWROUTE => Message::NewRoute(RouteMessage::parse(payload)?),
255            libc::RTM_DELROUTE => Message::DelRoute(RouteMessage::parse(payload)?),
256            libc::RTM_NEWRULE => Message::NewRule,
257            libc::RTM_DELRULE => Message::DelRule,
258            kind => Message::Other { kind },
259        };
260        Some(message)
261    }
262}

264/// One frame of a received datagram, before dump or event bookkeeping.
265#[derive(Debug)]
266pub(crate) enum Frame {
267    /// An rtnetlink message.
268    Message { seq: u32, message: Message },
269    /// `NLMSG_DONE`, the end of a multipart response.
270    Done { seq: u32 },
271    /// `NLMSG_ERROR`; `code` is zero for acknowledgments.
272    Error { seq: u32, code: i32 },
273    /// A frame to ignore (noop, overrun or malformed).
274    Skip,
275}
276
277/// Splits a received datagram into its netlink frames.
278///
279/// Iteration stops at the first frame whose length field is inconsistent
280/// with the remaining data.
281pub(crate) fn parse_frames(datagram: &[u8]) -> impl Iterator<Item = Frame> + '_ {
282    let mut data = datagram;
283    std::iter::from_fn(move || {
284        let header = wire::Header::parse(data)?;
285        let len = header.len as usize;
286        if len < wire::HEADER_LEN || len > data.len() {
287            return None;
288        }
289        let payload = &data[wire::HEADER_LEN..len];
290        data = data.get(wire::align(len)..).unwrap_or_default();
291        let frame = match header.kind as i32 {
292            libc::NLMSG_DONE => Frame::Done { seq: header.seq },
293            libc::NLMSG_ERROR => match wire::read_i32(payload, 0) {
294                Some(code) => Frame::Error {
295                    seq: header.seq,
296                    code,
297                },
298                None => Frame::Skip,
299            },
300            libc::NLMSG_NOOP | libc::NLMSG_OVERRUN => Frame::Skip,
301            _ => match Message::parse(header.kind, payload) {
302                Some(message) => Frame::Message {
303                    seq: header.seq,
304                    message,
305                },
306                None => Frame::Skip,
307            },
308        };
309        Some(frame)
310    })
311}

313/// Builds an `RTM_GETLINK` dump request.
314pub(crate) fn dump_links_request(seq: u32) -> Vec<u8> {
315    dump_request(libc::RTM_GETLINK, seq, &[0u8; 16])
316}

318/// Builds an `RTM_GETADDR` dump request covering both address families.
319pub(crate) fn dump_addresses_request(seq: u32) -> Vec<u8> {
320    dump_request(libc::RTM_GETADDR, seq, &[0u8; 8])
321}

323/// Builds an `RTM_GETROUTE` dump request.
324///
325/// The filter fields (main table, static protocol, unicast type) match what
326/// netwatch has always sent; the kernel ignores them for non-strict dumps
327/// and filters by family only.
328pub(crate) fn dump_routes_request(seq: u32, family: RouteFamily) -> Vec<u8> {
329    let mut rtmsg = [0u8; 12];
330    rtmsg[0] = family.family();
331    rtmsg[4] = libc::RT_TABLE_MAIN;
332    rtmsg[5] = libc::RTPROT_STATIC;
333    rtmsg[6] = libc::RT_SCOPE_UNIVERSE;
334    rtmsg[7] = libc::RTN_UNICAST;
335    dump_request(libc::RTM_GETROUTE, seq, &rtmsg)
336}

338/// Builds an `RTM_GETLINK` request for a single interface index.
339pub(crate) fn get_link_request(seq: u32, index: u32) -> Vec<u8> {
340    let mut ifinfomsg = [0u8; 16];
341    ifinfomsg[4..8].copy_from_slice(&(index as i32).to_ne_bytes());
342    let mut buf = Vec::with_capacity(wire::HEADER_LEN + ifinfomsg.len());
343    wire::push_header(
344        &mut buf,
345        libc::RTM_GETLINK,
346        libc::NLM_F_REQUEST as u16,
347        seq,
348        ifinfomsg.len(),
349    );
350    buf.extend_from_slice(&ifinfomsg);
351    buf
352}
353
354fn dump_request(kind: u16, seq: u32, fixed_header: &[u8]) -> Vec<u8> {
355    let mut buf = Vec::with_capacity(wire::HEADER_LEN + fixed_header.len());
356    let flags = (libc::NLM_F_REQUEST | libc::NLM_F_DUMP) as u16;
357    wire::push_header(&mut buf, kind, flags, seq, fixed_header.len());
358    buf.extend_from_slice(fixed_header);
359    buf
360}

362/// Parses a NUL-terminated attribute payload into a string.
363fn parse_string(payload: &[u8]) -> Option<String> {
364    let end = payload
365        .iter()
366        .position(|&b| b == 0)
367        .unwrap_or(payload.len());
368    String::from_utf8(payload[..end].to_vec()).ok()
369}
370
371/// Parses an address attribute payload according to the message family.
372fn parse_ip(family: u8, payload: &[u8]) -> Option<IpAddr> {
373    if family as i32 == libc::AF_INET {
374        let bytes: [u8; 4] = payload.get(..4)?.try_into().ok()?;
375        Some(IpAddr::from(bytes))
376    } else if family as i32 == libc::AF_INET6 {
377        let bytes: [u8; 16] = payload.get(..16)?.try_into().ok()?;
378        Some(IpAddr::from(bytes))
379    } else {
380        None
381    }
382}

391    /// Builds a response frame the way the kernel would.
392    fn build_frame(kind: u16, seq: u32, payload: &[u8]) -> Vec<u8> {
393        let mut buf = Vec::new();
394        wire::push_header(&mut buf, kind, libc::NLM_F_MULTI as u16, seq, payload.len());
395        buf.extend_from_slice(payload);
396        buf.resize(wire::align(buf.len()), 0);
397        buf
398    }
399
400    fn push_attr(buf: &mut Vec<u8>, kind: u16, payload: &[u8]) {
401        let len = (4 + payload.len()) as u16;
402        buf.extend_from_slice(&len.to_ne_bytes());
403        buf.extend_from_slice(&kind.to_ne_bytes());
404        buf.extend_from_slice(payload);
405        buf.resize(wire::align(buf.len()), 0);
406    }
